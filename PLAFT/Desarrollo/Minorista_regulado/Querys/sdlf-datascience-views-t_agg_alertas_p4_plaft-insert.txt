select a.cod_cuc,
sum(mto_dolares) as monto_usd_trx, count(1) as cantidad_trx
from wrk_aml_operaciones a
cross join wrk_dim_tiempo t
where a.periodo between t.lcodmesm12 and t.lcodmesm01
group by 1
),
wrk_mga_rfm_regulares as (
select a.cod_cuc,
monto_usd_trx,
coalesce(a.cantidad_trx/a.cantidad_trx,0) as p_lugar_operativa_a
from wrk_pre_rfm_regulares a
),
wrk_trx_tipo_operacion as (
select
a.cod_cli,
count(1) as cantidad_trx_tipo,
SUM(mto_dolares) AS monto_usd_trx
from trx_egreso_ingreso a
inner join mst_plaft_trx_operaciones b
on a.cod_trx_orig = b.cod_trx
group by a.cod_cli
),
wrk_trx_tipo_operacion_r100 as (
select
a.cod_cli,
count(1) as cantidad_trx_riesgo,
SUM(mto_dolares) AS monto_usd_trx
from trx_egreso_ingreso a
inner join mst_plaft_trx_operaciones b
on a.cod_trx_orig = b.cod_trx
where b.cod_riesgo = '100'
group by a.cod_cli
),
wrk_p_tipo_operacion_a as (
select
t.cod_cli,
coalesce(
r.cantidad_trx_riesgo * 1.0 / t.cantidad_trx_tipo,
0
) as p_tipo_operacion_a,
t.monto_usd_trx
from wrk_trx_tipo_operacion t
left join wrk_trx_tipo_operacion_r100 r
on t.cod_cli = r.cod_cli
),
wrk_mga_umbrales_rfm as (
select max(case when rn <= 0.5 * cnt then monto_usd_trx else 0 end) as valor1_m,
max(case when rn <= 0.8 * cnt then monto_usd_trx else 0 end) as valor1_a,
0.5 as valor2_m,
0.8 as valor2_a
from (
select
monto_usd_trx,
row_number() over (order by monto_usd_trx asc) as rn,
count(*)     over () as cnt
from wrk_mga_rfm_regulares
) t
),
wrk_riesgo_lugar_operativa as (
select t1.cod_cuc as codunico,
case 
 when t1.p_lugar_operativa_a>=t2.valor2_a and t1.monto_usd_trx>=t2.valor1_a  then 100
 when t1.p_lugar_operativa_a>=t2.valor2_m and t1.monto_usd_trx>=t2.valor1_m  then 50
 else 0
end as v13_lugar_operativa_riesgo
from wrk_mga_rfm_regulares t1
cross join wrk_mga_umbrales_rfm t2
),
wrk_riesgo_tipo_operacion as (
select t1.cod_cli as codunico,
CASE 
WHEN P_TIPO_OPERACION_A>=VALOR2_A AND MONTO_USD_TRX>=VALOR1_A  THEN 100
WHEN P_TIPO_OPERACION_A>=VALOR2_M AND MONTO_USD_TRX>=VALOR1_M  THEN 50
ELSE 0
END AS V16_CANAL_OPERACION_RIESGO
from wrk_p_tipo_operacion_a t1
cross join wrk_mga_umbrales_rfm t2
),
wrk_trx_tipo_ubigeo as (
select
a.cod_cli,
count(1) as cantidad_trx_riesgo,
SUM(mto_dolares) AS monto_usd_trx
from trx_egreso_ingreso a
inner join mst_plaft_trx_operaciones b
on a.cod_trx_orig = b.cod_trx
where b.cod_riesgo = '100'
group by a.cod_cli
),
V12_LUGAR_RESIDENCIA AS (  
SELECT
t1.cod_cli,
COALESCE(t2.ubg_codriesgo, '100') AS v12_lugar_residencia_riesgo          
FROM trx_egreso_ingreso t1              
LEFT JOIN mst_plaft_dim_ubigeo t2              
ON t1.cod_ubigeo_opera = t2.cod_ubg_departamento              
AND CASE WHEN t1.cod_ubigeo_opera <> '15' THEN '' END = COALESCE(t2.ubg_distrito, '')
),
consolidado_sunat AS (
SELECT
b.cod_cuc AS codunico,
ccr.des_actividad AS descripcion_ciiu_v4,
ROW_NUMBER() OVER (
PARTITION BY t.numruc 
ORDER BY t.fecvigenciarele DESC
) AS rn
FROM base_cucibk b
INNER JOIN e_perm_aws.t_consolidado_sunat t
ON b.key_value = t.numruc
CROSS JOIN wrk_dim_tiempo d
LEFT JOIN e_perm_aws.t_mst_catalogo_ciiu_rsk CCR
ON T.ciiuv4 = CCR.cod_ciiu
),
wkr_tipo_cambio_captaciones AS (
select codmes, tipo_cambio, cod_moneda_origen
from (
select to_char(fecha_cambio,'yyyymm') codmes, cod_tipo_cambio,
round(tipo_cambio,3) tipo_cambio, cod_moneda_origen,fecha_cambio,
row_number() over(partition by date_trunc('month', fecha_cambio), cod_moneda_origen order by fecha_cambio desc) r
from e_perm_aws.t_dim_tipo_cambio_rsk a
cross JOIN wrk_dim_tiempo t
where to_char(fecha_cambio,'yyyymm') BETWEEN t.lcodmesm14 AND t.codmes
and cod_moneda_origen in ('USD','EUR')
and cod_tipo_cambio = '1002'
) tc
where tc.r=1
order by 1 desc
),
wrk_trx_captaciones as (
select
date_format(fechinievento,'%Y%m') as periodo,
cuc_num,
SUM(CASE WHEN M.SIGNOMONTO = '+' AND M.MONEDACD = 'PSS' THEN M.MONTOTRANSACCION
WHEN M.SIGNOMONTO = '+' AND M.MONEDACD = 'USD' THEN M.MONTOTRANSACCION*b.tipo_cambio
WHEN M.SIGNOMONTO = '+' AND M.MONEDACD = 'EUR' THEN M.MONTOTRANSACCION*b.tipo_cambio
ELSE 0 END) AS ABONOS_SOLARIZADO
FROM e_perm_aws.t_fct_transacciones_captaciones m
cross join wrk_dim_tiempo t
left join wkr_tipo_cambio_captaciones b on date_format(m.fechinievento,'%Y%m') = b.codmes and m.monedacd = b.cod_moneda_origen
WHERE m.fechinievento between t.dlcodmesm13 and t.dlcodmesm00
and m.signomonto in ('+','-') and cuc_num is not null
and cuc_num <> coalesce(cuc_num_ord_ben, 'abc')
group by 1,2
),
cantidad_monto_abonos_total AS (
SELECT
w.CUC_NUM AS codunico,
SUM(CASE WHEN w.periodo BETWEEN t.lcodmesm03 AND t.lcodmesm01 
THEN w.abonos_solarizado 
ELSE 0 
END) AS TRX_MONTO_ABONOS_1A3M_TOTAL,
SUM(CASE WHEN w.periodo BETWEEN t.lcodmesm06 AND t.lcodmesm04 
THEN w.abonos_solarizado 
ELSE 0 
END) AS TRX_MONTO_ABONOS_4A6M_TOTAL,
SUM(CASE WHEN w.periodo BETWEEN t.lcodmesm09 AND t.lcodmesm04 
THEN w.abonos_solarizado 
ELSE 0 
END) AS TRX_MONTO_ABONOS_4A9M_TOTAL,
SUM(CASE WHEN w.periodo BETWEEN t.lcodmesm06 AND t.lcodmesm01 
THEN w.abonos_solarizado 
ELSE 0 
END) AS TRX_MONTO_ABONOS_1A6M_TOTAL,
SUM(CASE WHEN w.periodo BETWEEN t.lcodmesm12 AND t.lcodmesm07 
THEN w.abonos_solarizado 
ELSE 0 
END) AS TRX_MONTO_ABONOS_7A12M_TOTAL
FROM wrk_trx_captaciones w
CROSS JOIN wrk_dim_tiempo t
GROUP BY w.CUC_NUM
),
ratio_monto_abonos AS (
SELECT
codunico,
COALESCE(CAST(TRX_MONTO_ABONOS_1A3M_TOTAL * 1.000000
/ NULLIF(TRX_MONTO_ABONOS_4A6M_TOTAL, 0)
AS DECIMAL(38,6)),DECIMAL '0.000000' ) AS TRX_MONTO_ABONOS_RATIO_1A3M_4A6M_TOTAL,
COALESCE(CAST(TRX_MONTO_ABONOS_1A3M_TOTAL * 1.000000
/ NULLIF(TRX_MONTO_ABONOS_4A9M_TOTAL, 0)
AS DECIMAL(38,6)), DECIMAL '0.000000') AS TRX_MONTO_ABONOS_RATIO_1A3M_4A9M_TOTAL,
COALESCE(CAST(TRX_MONTO_ABONOS_1A6M_TOTAL * 1.000000
/ NULLIF(TRX_MONTO_ABONOS_7A12M_TOTAL, 0)
AS DECIMAL(38,6)), DECIMAL '0.000000') AS TRX_MONTO_ABONOS_RATIO_1A6M_7A12M_TOTAL
FROM cantidad_monto_abonos_total
)
SELECT distinct
t.codmes AS COD_MES,
CAST(bc.codunico AS VARCHAR) AS COD_CLI,
COALESCE(CAST(mtri.ING_TIENDA_RIESGO_BAJO AS DECIMAL(38,6)), 0) AS MTO_ING_TNDA_RSG_BAJO_12M,
COALESCE(CAST(mtri.ING_TIENDA_RIESGO_MEDIO AS DECIMAL(38,6)), 0) AS MTO_ING_TNDA_RSGMEDIO_12M,
COALESCE(CAST(mtri.ING_TIENDA_RIESGO_ALTO AS DECIMAL(38,6)), 0) AS MTO_ING_TNDA_RSG_ALTO_12M,
COALESCE(CAST(mtri.Q_TIENDA_RIESGO_BAJO AS INTEGER), 0) AS CNT_TIENDA_RSG_BAJO_12M,
COALESCE(CAST(mtri.Q_TIENDA_RIESGO_MEDIO AS INTEGER), 0) AS CNT_TIENDA_RSG_MEDIO_12M,
COALESCE(CAST(mtri.Q_TIENDA_RIESGO_ALTO AS INTEGER), 0) AS CNT_TIENDA_RSG_ALTO_12M,
COALESCE(CAST(mcri.ING_CANAL_RIESGO_BAJO AS DECIMAL(38,6)), 0) AS MTO_ING_CNL_RSG_BAJO_12M,
COALESCE(CAST(mcri.ING_CANAL_RIESGO_MEDIO AS DECIMAL(38,6)), 0) AS MTO_ING_CNL_RSG_MEDIO_12M,
COALESCE(CAST(mcri.ING_CANAL_RIESGO_ALTO AS DECIMAL(38,6)), 0)  AS MTO_ING_CNL_RSG_ALTO_12M,
COALESCE(CAST(mcri.Q_CANAL_RIESGO_BAJO AS INTEGER), 0) AS CNT_CANAL_RSG_BAJO_12M,
COALESCE(CAST(mcri.Q_CANAL_RIESGO_MEDIO AS INTEGER), 0) AS CNT_CANAL_RSG_MEDIO_12M,
COALESCE(CAST(mcri.Q_CANAL_RIESGO_ALTO AS INTEGER), 0) AS CNT_CANAL_RSG_ALTO_12M,
COALESCE(CAST(mtder.ING_DEL_EXT_RIESGO_ALTO AS DECIMAL(38,6)), 0) AS MTO_ING_DELEXTRSGALTO_12M,
COALESCE(CAST(mtder.ING_DEL_EXT_TOTAL AS DECIMAL(38,6)), 0) AS MTO_ING_DELEXTTOT_12M,
COALESCE(CAST(mtder.Q_DEL_EXT_RIESGO_ALTO AS INTEGER), 0) AS CNT_TRX_DELEXTRSGALTO_12M,
COALESCE(CAST(mtder.Q_DEL_EXT_TOTAL AS INTEGER), 0) AS CNT_TRX_DELEXTTOT_12M,
COALESCE(CAST(mtder.ING_AL_EXT_RIESGO_ALTO AS DECIMAL(38,6)), 0) AS MTO_AL_EXTRSGALTOT_12M,
COALESCE(CAST(mtder.ING_AL_EXT_TOTAL AS DECIMAL(38,6)), 0) AS MTO_AL_EXT_TOT_12M,
COALESCE(CAST(mtder.Q_AL_EXT_RIESGO_ALTO AS INTEGER), 0) AS CNT_TRX_ALEXTRSGALTO,
COALESCE(CAST(mtder.Q_AL_EXT_TOTAL AS INTEGER), 0) AS CNT_AL_EXT_TOT_12M,
CAST(COALESCE(
CAST(mtder.ing_del_ext_riesgo_alto AS DECIMAL(38,6))/ NULLIF(CAST(mtder.ing_del_ext_total AS DECIMAL(38,6)), 0),0)
AS DECIMAL(38,6)) AS RAT_ING_DL_XT_LT_TOT_12M,
CAST(COALESCE(
CAST(mtder.q_del_ext_riesgo_alto AS DECIMAL(38,6))/ NULLIF(CAST(mtder.q_del_ext_total AS DECIMAL(38,6)), 0),0)
AS DECIMAL(38,6)) AS RAT_TRX_INGEXTRSGALTO_12M,
CAST(COALESCE(
CAST(mtder.ing_al_ext_riesgo_alto AS DECIMAL(38,6))/ NULLIF(CAST(mtder.ing_al_ext_total AS DECIMAL(38,6)), 0),0)
AS DECIMAL(38,6)) AS RAT_MTO_AL_XT_LT_12M,
CAST(COALESCE(
CAST(mtder.q_al_ext_riesgo_alto AS DECIMAL(38,6))/ NULLIF(CAST(mtder.q_al_ext_total AS DECIMAL(38,6)), 0),0)
AS DECIMAL(38,6)) AS RAT_CNT_ING_AL_XT_LT_TOT,
COALESCE(CAST(rlo.v13_lugar_operativa_riesgo AS INTEGER), 0) AS COD_V13_LUGAR_OP_RSG_12M,
COALESCE(CAST(rto.V16_CANAL_OPERACION_RIESGO AS INTEGER), 0) AS COD_V16_CANAL_OP_RSG_12M,
CAST(cs.descripcion_ciiu_v4 AS VARCHAR) AS DESC_CIIU_V4,
COALESCE(CAST(rma.TRX_MONTO_ABONOS_RATIO_1A3M_4A6M_TOTAL AS DECIMAL(38,6)), 0) AS RAT_TRX_MNTABONTOT_13_46,
COALESCE(CAST(rma.TRX_MONTO_ABONOS_RATIO_1A3M_4A9M_TOTAL AS DECIMAL(38,6)), 0) AS RAT_TRX_MNTABONTOT_13_49,
COALESCE(CAST(rma.TRX_MONTO_ABONOS_RATIO_1A6M_7A12M_TOTAL AS DECIMAL(38,6)), 0) AS RAT_TRX_MNTABONTOT_16_712,
COALESCE(TRY_CAST(lr.v12_lugar_residencia_riesgo AS INTEGER), 0) AS COD_V12_LUGAR_RSDN_RSG,
COALESCE(TRY_CAST(mtder.v11_pais_operativa_riesgo AS INTEGER), 0) AS COD_V11_PAIS_OP_RSG,
(localtimestamp - interval '5' hour) as ts_carga,
t.codmes AS p_codmes
FROM base_clientes bc
cross join wrk_dim_tiempo t
LEFT JOIN mtz_trx_canal_riesgo_ingreso mcri
ON bc.codunico = mcri.codunico
LEFT JOIN v12_lugar_residencia lr
ON bc.codunico = lr.cod_cli
LEFT JOIN mtz_trx_tienda_riesgo_ingreso mtri
ON bc.codunico = mtri.codunico
LEFT JOIN MTZ_TRX_DEL_EXT_RIESGO mtder
ON bc.codunico = mtder.codunico
left join wrk_riesgo_lugar_operativa rlo
on bc.codunico = rlo.codunico
LEFT JOIN wrk_riesgo_tipo_operacion rto
on bc.codunico = rto.codunico
LEFT JOIN ratio_monto_abonos rma
ON bc.codunico = rma.codunico
LEFT JOIN consolidado_sunat cs
ON bc.codunico = cs.codunico AND cs.rn = 1
;
